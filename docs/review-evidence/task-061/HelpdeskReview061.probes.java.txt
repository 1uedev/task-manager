    private static class ReviewConnection implements MailClient.Connection {
        final long validity;final byte[] raw;
        Runnable listHook=()->{};
        int fetches;boolean seen;
        ReviewConnection(long validity,byte[] raw){this.validity=validity;this.raw=raw;}
        public long uidValidity(){return validity;}
        public List<Long> uids(long last,boolean unseen,int limit){if(!unseen && limit!=Integer.MAX_VALUE)listHook.run();return last<1?List.of(1L):List.of();}
        public long size(long uid){return raw.length;}
        public byte[] fetch(long uid){fetches++;return raw;}
        public ParsedMail header(long uid){try{return ParsedMail.headerOnly(new jakarta.mail.internet.MimeMessage(Session.getInstance(new Properties()),new java.io.ByteArrayInputStream(raw)));}catch(Exception e){throw new AssertionError(e);}}
        public void markSeen(long uid){seen=true;}
        public void close(){}
    }

    @Test
    void probeKnownOriginCanBeMarkedSeenAfterRestartOrSettingsCommit() {
        for(boolean settings:List.of(false,true)) {
            Setup s=setup();Setup other=setup();byte[] raw=scripted(s,"known-origin","");
            new ScriptedMailbox(100).put(1,raw).intake(context,5).fetch(s.mailbox());
            jdbc.update("update helpdesk_mailbox set last_uid=0 where id=?",s.mailbox());
            ReviewConnection old=new ReviewConnection(100,raw);
            old.listHook=()->{
                if(settings)changeAccount(s,other);
                else new ScriptedMailbox(200).put(1,raw).intake(context,5).fetch(s.mailbox());
            };
            probe((m,p)->old,transactions).fetch(s.mailbox());

            assertEquals(settings?0L:1L,jdbc.queryForObject("select last_uid from helpdesk_mailbox where id=?",Long.class,s.mailbox()));
            System.out.println("E1 known origin: stale markSeen after "+(settings?"settings":"folder restart")+" commit; DB outcome/cursor unchanged");
        }
    }

    @Test

        Setup s=setup();byte[] raw=scripted(s,"after-record","");

    private void awaitMailboxWait() {
        long end=System.nanoTime()+java.util.concurrent.TimeUnit.SECONDS.toNanos(10);
        while(System.nanoTime()<end){
            Integer waits=jdbc.queryForObject("select count(*) from pg_stat_activity where datname=current_database() and wait_event_type='Lock' and query like '%helpdesk_mailbox%'",Integer.class);
            if(waits>0)return;
            try{Thread.sleep(20);}catch(InterruptedException e){throw new AssertionError(e);}
        }
        throw new AssertionError("no mailbox lock wait observed");
    }

    @Test
    void probeMailboxLockSerializesRestartAndSettingsWithoutDeadlock() throws Exception {
        for(boolean settings:List.of(false,true)){
            Setup s=setup();Setup other=setup();

            var tx=new TransactionTemplate(transactions);var locked=new java.util.concurrent.CountDownLatch(1);var release=new java.util.concurrent.CountDownLatch(1);
            Future<?> holder=POOL.submit(()->tx.execute(status->{assertTrue(reviewRepository.lockCurrent(s.mailbox(),1,100));locked.countDown();hold(release);return null;}));
            Future<?> contender=null;

                hold(locked);
