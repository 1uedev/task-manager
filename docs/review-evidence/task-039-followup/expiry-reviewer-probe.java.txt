    private boolean reviewWaitsFor(java.util.concurrent.atomic.AtomicInteger waiter,
            java.util.concurrent.atomic.AtomicInteger holder, String statementPart) throws InterruptedException {
        long deadline = System.nanoTime() + TimeUnit.SECONDS.toNanos(4);
        while (System.nanoTime() < deadline) {
            if (waiter.get() > 0 && holder.get() > 0 && Boolean.TRUE.equals(jdbc.queryForObject("""
                    select exists(select 1 from pg_stat_activity
                    where datname = current_database() and pid = ?
                    and ? = any(pg_blocking_pids(pid)) and query ilike ?)
                    """, Boolean.class, waiter.get(), holder.get(), "%" + statementPart + "%"))) {
                return true;
            }
            Thread.sleep(10);
        }
        return false;
    }

    @Test
    void reviewProbeExpiryDuringSessionLockWaitMustNotCommitABusinessEffect() throws Exception {
        Person chef = bootstrap("chef");
        Person reader = account("reader", false);
        grantTicketRead(chef, reader);
        AccessContext reading = ctx(reader);
        CLOCK.advance(Duration.ofMinutes(59).plusSeconds(59));
        var holderPid = new java.util.concurrent.atomic.AtomicInteger();
        var waiterPid = new java.util.concurrent.atomic.AtomicInteger();
        jdbc.update("create table review_task039_effect(id int primary key)");
        try {
            Held<Boolean> shortening = hold(() -> {
                holderPid.set(jdbc.queryForObject("select pg_backend_pid()", Integer.class));
                duration.change("test", 1);
                return true;
            });
            Future<Boolean> business = POOL.submit(() -> {
                try {
                    return inTransaction(() -> {
                        waiterPid.set(jdbc.queryForObject("select pg_backend_pid()", Integer.class));
                        permissions.require(reading, companyScope(), "ticket.read", Set.of(), Set.of());
                        jdbc.update("insert into review_task039_effect values (1)");
                        return true;
                    });
                } catch (PermissionDenied denied) {
                    return false;
                }
            });
            boolean exactWait = reviewWaitsFor(waiterPid, holderPid, "for share of s");
            try {
                assertTrue(exactWait, "the check must wait for this exact shortening transaction");
                assertFalse(business.isDone());
                CLOCK.advance(Duration.ofSeconds(2));
            } finally {
                shortening.finish();
            }
            boolean allowed = business.get(15, TimeUnit.SECONDS);
            boolean fresh = permissions.inspect(reading, companyScope(), "ticket.read", Set.of()).allowed();
            int effects = jdbc.queryForObject("select count(*) from review_task039_effect", Integer.class);
            System.out.println("REVIEW_EXPIRY exactWait=" + exactWait + " oldCheckAllowed=" + allowed
                + " freshAllowed=" + fresh + " committedEffects=" + effects);
            assertEquals(0, effects, "A check released after expiry must not commit a business effect");
        } finally {
            jdbc.update("drop table review_task039_effect");
        }
    }
