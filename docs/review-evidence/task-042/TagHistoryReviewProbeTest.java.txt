    record Response(int status, String body, Map<String, List<String>> headers) {
        String header(String name) {
            return headers.entrySet().stream().filter(e -> e.getKey().equalsIgnoreCase(name))
                .flatMap(e -> e.getValue().stream()).findFirst().orElse(null);
        }

        JsonNode json() {
            return JSON.readTree(body);
        }

        String code() {
            return json().get("code").asString();
        }

        List<String> violations() {
            return StreamSupport.stream(json().get("violations").spliterator(), false).map(JsonNode::asString).toList();
        }

        JsonNode fields() {
            return json().get("fields");
        }
    }

    final class Browser {
        private final HttpClient http = HttpClient.newHttpClient();
        final Map<String, String> cookies = new LinkedHashMap<>();

        Response send(String method, String path, Object body, String ifMatch) {
            try {
                HttpRequest.Builder request = HttpRequest.newBuilder(URI.create("http://localhost:" + port + path));
                if (body != null) {
                    request.header("Content-Type", "application/json");
                    request.method(method, HttpRequest.BodyPublishers.ofString(body instanceof String s ? s : JSON.writeValueAsString(body)));
                } else {
                    request.method(method, HttpRequest.BodyPublishers.noBody());
                }
                if (ifMatch != null) {
                    request.header("If-Match", ifMatch);
                }
                if (!cookies.isEmpty()) {
                    List<String> pairs = new ArrayList<>();
                    cookies.forEach((name, value) -> pairs.add(name + "=" + value));
                    request.header("Cookie", String.join("; ", pairs));
                }
                String csrf = cookies.get("__Host-tm-csrf");
                if (csrf != null) {
                    request.header("X-CSRF-Token", csrf);
                }
                HttpResponse<String> response = http.send(request.build(), HttpResponse.BodyHandlers.ofString());
                for (String cookie : response.headers().allValues("set-cookie")) {
                    String pair = cookie.substring(0, cookie.indexOf(';') < 0 ? cookie.length() : cookie.indexOf(';'));
                    String name = pair.substring(0, pair.indexOf('='));
                    if (cookie.toLowerCase().contains("max-age=0")) {
                        cookies.remove(name);
                    } else {
                        cookies.put(name, pair.substring(pair.indexOf('=') + 1));
                    }
                }
                return new Response(response.statusCode(), response.body(), response.headers().map());
            } catch (java.io.IOException | InterruptedException e) {
                throw new IllegalStateException(e);
            }
        }

        Response get(String path) {
            return send("GET", path, null, null);
        }
    }

    private Person person(AdministrationRight... rights) {
        var issued = accounts.create("test", "tags-" + COUNTER.incrementAndGet(), null);

        return new Person(browser, issued.account().id());
    }

    /**
     * Grants rules in a group of their own, for one project or, with {@code null}, for every project of the company: an
     * action key such as {@code ticket.read}, or a field rule {@code READ:title}, {@code WRITE:title}, {@code DENY-READ:title}
     * for the resource type {@code ticket}; {@code *} stands for all five fields, and {@code tags} is the field of the tags.
     */
    private void grant(UUID account, UUID project, String... rules) {
        new TransactionTemplate(transactions).executeWithoutResult(status -> {
            UUID companyId = companies.find().orElseGet(companies::create);
            jdbc.update("insert into permissions_revision(company_id) values (?) on conflict do nothing", companyId);
            UUID group = UUID.randomUUID();
            jdbc.update("insert into permissions_group(company_id, id, name) values (?, ?, ?)", companyId, group, "fixture-" + group);
            for (String rule : rules) {
                if (!rule.contains(":")) {
                    jdbc.update("insert into permissions_action(company_id, group_id, project_id, action_key) values (?, ?, ?, ?)",
                        companyId, group, project, rule);
                    continue;
                }
                String[] parts = rule.split(":");
                boolean deny = parts[0].startsWith("DENY-");
                String operation = deny ? parts[0].substring(5) : parts[0];
                for (String field : parts[1].equals("*") ? ALL_FIELDS : List.of(parts[1])) {
                    jdbc.update("""
                            insert into permissions_field(company_id, group_id, project_id, resource_type, field_key, operation, effect)
                            values (?, ?, ?, 'ticket', ?, ?, ?)
                            """, companyId, group, project, field, operation, deny ? "DENY" : "ALLOW");
                }
            }
            jdbc.update("insert into permissions_membership(company_id, group_id, user_id) values (?, ?, ?)", companyId, group, account);
            jdbc.update("update permissions_revision set revision = revision + 1 where company_id = ?", companyId);
        });
    }

    }

    // ---------------------------------------------------------------- the company catalog
    /** Runs the statements in a transaction of its own and keeps it open until the returned action releases it. */
    private Runnable hold(String... statements) throws Exception {
        CountDownLatch locked = new CountDownLatch(1);
        CountDownLatch release = new CountDownLatch(1);
        Future<?> holder = POOL.submit(() -> new TransactionTemplate(transactions).executeWithoutResult(status -> {
            for (String statement : statements) {
                jdbc.execute(statement);
            }
            locked.countDown();
            try {
                assertTrue(release.await(60, TimeUnit.SECONDS));
            } catch (InterruptedException interrupted) {
                Thread.currentThread().interrupt();
            }
        }));
        assertTrue(locked.await(15, TimeUnit.SECONDS));
        return () -> {
            release.countDown();
            try {
                holder.get(60, TimeUnit.SECONDS);
            } catch (Exception e) {
                throw new IllegalStateException(e);
            }
        };
    }

    private boolean waitsAt(String statementPart) throws InterruptedException {
        long deadline = System.nanoTime() + TimeUnit.SECONDS.toNanos(10);
        while (System.nanoTime() < deadline) {
            if (Boolean.TRUE.equals(jdbc.queryForObject("""
                    select exists(select 1 from pg_stat_activity
                    where datname = current_database() and pid <> pg_backend_pid()
                    and query ilike ? and cardinality(pg_blocking_pids(pid)) > 0)
                    """, Boolean.class, "%" + statementPart + "%"))) {
                return true;
            }
            Thread.sleep(10);
        }
        return false;
    }

    private String history(Project project, long number) {
        return "/api/v1/projects/" + project.id() + "/tickets/" + number + "/history";
    }

    private UUID ticketId(Project project, long number) {
        return jdbc.queryForObject("select id from ticket where project_id = ? and number = ?", UUID.class, project.id(), number);
    }

    /** The history as lines: version, then the field changes, tags as +id or -id. */
    private static List<String> lines(Response response) {
        assertEquals(200, response.status(), response.body());
        List<String> lines = new ArrayList<>();
        for (JsonNode entry : response.json()) {
            StringBuilder line = new StringBuilder(String.valueOf(entry.get("version").asLong()));
            for (JsonNode change : entry.get("changes")) {
                String field = change.get("field").asString();
                if (field.equals("tags")) {
                    line.append(change.get("from").isNull() ? " +" + change.get("to").asString() : " -" + change.get("from").asString());
                } else {
                    line.append(' ').append(field);
                }
            }
            lines.add(line.toString());
        }
        return lines;
    }

    private void bump() {
        jdbc.update("update permissions_revision set revision = revision + 1");
    }

    @Test
    void reviewSeveralTagChangesShareOneVersionAndArchivedTagsAreRecordedOnRemoval() {
        Person worker = person();
        Project project = project(worker);
        grant(worker.id(), project.id(), TICKETS);
        grant(worker.id(), project.id(), "tag.create");
        Person admin = person(AdministrationRight.TAGS_MANAGE);
        UUID company = tag(admin, "/api/v1/tags", "Firma " + COUNTER.incrementAndGet());
        long number = ticket(worker, project, "Viele Tags");
        UUID a = tag(worker, catalog(project), "A");
        UUID b = tag(worker, catalog(project), "B");

        assertEquals(204, post(worker, ticketTags(project, number), Map.of("tagId", a.toString())).status());
        assertEquals(204, post(worker, ticketTags(project, number), Map.of("tagId", b.toString())).status());
        assertEquals(204, post(worker, ticketTags(project, number), Map.of("tagId", company.toString())).status());
        assertEquals(200, patch(admin, "/api/v1/tags/" + company, Map.of("archived", true), "\"1\"").status());
        assertEquals(204, delete(worker, ticketTags(project, number) + "/" + company).status());
        assertEquals(204, delete(worker, ticketTags(project, number) + "/" + UUID.randomUUID()).status());
        List<String> lines = lines(worker.get(history(project, number)));
        assertEquals(List.of("1 +" + a, "1 +" + b, "1 +" + company, "1 -" + company), lines.subList(1, lines.size()), lines.toString());
    }

    @Test
    void reviewProjectBoundariesAndMissingReadActionHideTheHistory() {
        Person worker = person();
        Person stranger = person();
        Person outsider = person();
        Person editor = person();
        Project project = project(worker, editor);
        Project foreign = project(stranger);
        grant(worker.id(), project.id(), TICKETS);
        grant(stranger.id(), foreign.id(), TICKETS);
        grant(outsider.id(), null, TICKETS);
        grant(editor.id(), project.id(), "ticket.edit", "READ:tags", "WRITE:tags", "READ:*");
        long number = ticket(worker, project, "Grenzen");
        UUID tag = tag(person(AdministrationRight.TAGS_MANAGE), "/api/v1/tags", "Grenze " + COUNTER.incrementAndGet());
        assertEquals(204, post(editor, ticketTags(project, number), Map.of("tagId", tag.toString())).status());

        assertProblem(stranger.get(history(project, number)), 404, "project.not_found");
        assertProblem(outsider.get(history(project, number)), 404, "project.not_found");
        assertProblem(editor.get(history(project, number)), 404, "project.not_found");
        assertTrue(lines(worker.get(history(project, number))).contains("1 +" + tag), "the edit-only change is recorded");
    }

    @Test
    void reviewRevokedOrDeniedTagReadRuleHidesTheEntriesAtOnce() {
        Person worker = person();
        Person viewer = person();
        Person denied = person();
        Project project = project(worker, viewer, denied);
        grant(worker.id(), project.id(), TICKETS);
        grant(worker.id(), project.id(), "tag.create");
        grant(viewer.id(), project.id(), "ticket.read", "READ:*");
        grant(viewer.id(), project.id(), "READ:tags");
        grant(denied.id(), project.id(), "ticket.read", "READ:*", "READ:tags");
        grant(denied.id(), null, "DENY-READ:tags");
        long number = ticket(worker, project, "Entzug");
        UUID tag = tag(worker, catalog(project), "Entzug");
        assertEquals(204, post(worker, ticketTags(project, number), Map.of("tagId", tag.toString())).status());

        assertTrue(lines(viewer.get(history(project, number))).contains("1 +" + tag));
        assertFalse(lines(denied.get(history(project, number))).contains("1 +" + tag), "a deny rule wins");
        new TransactionTemplate(transactions).executeWithoutResult(status -> {
            jdbc.update("""
                    delete from permissions_membership m using permissions_field f
                    where m.group_id = f.group_id and m.user_id = ? and f.field_key = 'tags'
                    """, viewer.id());
            bump();
        });
        assertEquals(1, lines(viewer.get(history(project, number))).size(), "after the revocation only the creation remains");
    }

    @Test
    void reviewAFailedHistoryWriteRollsTheAssignmentBack() throws Exception {
        Person worker = person();
        Project project = project(worker);
        grant(worker.id(), project.id(), TICKETS);
        grant(worker.id(), project.id(), "tag.create");
        long number = ticket(worker, project, "Zeitüberschreitung");
        UUID tag = tag(worker, catalog(project), "Warten");
        UUID ticket = ticketId(project, number);
        Runnable release = hold("select id from ticket where id = '" + ticket + "' for update");
        try {
            Response busy = post(worker, ticketTags(project, number), Map.of("tagId", tag.toString()));
            assertProblem(busy, 503, "tag.busy");
            assertEquals("1", busy.header("Retry-After"));
        } finally {
            release.run();
        }
        assertEquals(0, jdbc.queryForObject("select count(*) from tag_ticket where ticket_id = ?", Integer.class, ticket));
        assertEquals(0, jdbc.queryForObject("select count(*) from tag_audit_event where operation = 'tag.assigned' and target_id = ?",
            Integer.class, ticket));
        assertEquals(1, lines(worker.get(history(project, number))).size());
    }

    @Test
    void reviewAnAssignmentThatWaitsForAFieldChangeTakesTheNewVersionAndALaterTime() throws Exception {
        Person worker = person();
        Project project = project(worker);
        grant(worker.id(), project.id(), TICKETS);
        grant(worker.id(), project.id(), "tag.create");
        long number = ticket(worker, project, "Reihenfolge");
        UUID tag = tag(worker, catalog(project), "Später");
        UUID ticket = ticketId(project, number);
        CountDownLatch locked = new CountDownLatch(1);
        CountDownLatch release = new CountDownLatch(1);
        Future<?> change = POOL.submit(() -> new TransactionTemplate(transactions).executeWithoutResult(status -> {
            jdbc.queryForList("select id from ticket where id = ? for update", ticket);
            locked.countDown();
            try {
                assertTrue(release.await(30, TimeUnit.SECONDS));
            } catch (InterruptedException interrupted) {
                Thread.currentThread().interrupt();
            }
            jdbc.update("update ticket set version = version + 1, title = 'Neu' where id = ?", ticket);
            jdbc.update("""
                    insert into ticket_change (ticket_id, version, occurred_at, actor_account_id, field, old_value, new_value)
                    select id, version, clock_timestamp(), ?, 'title', 'Reihenfolge', 'Neu' from ticket where id = ?
                    """, worker.id(), ticket);
        }));
        assertTrue(locked.await(15, TimeUnit.SECONDS));
        Future<Response> assign = POOL.submit(() -> post(worker, ticketTags(project, number), Map.of("tagId", tag.toString())));
        assertTrue(waitsAt("select version from ticket"), "the assignment waits at the ticket row");
        Thread.sleep(500);
        release.countDown();
        change.get(30, TimeUnit.SECONDS);
        assertEquals(204, assign.get(30, TimeUnit.SECONDS).status());

        Response seen = worker.get(history(project, number));
        assertEquals(List.of("1", "2", "2 +" + tag), lines(seen).stream().map(l -> l.startsWith("1 ") ? "1" : l.equals("2 title") ? "2" : l).toList());
        Instant field = Instant.parse(seen.json().get(1).get("occurredAt").asString());
        Instant tagged = Instant.parse(seen.json().get(2).get("occurredAt").asString());
        assertFalse(tagged.isBefore(field), "the later entry must not carry an earlier time: field " + field + ", tag " + tagged);
    }

    @Test
    void reviewConcurrentFieldAndTagChangesNeitherDeadlockNorMixUpTheHistory() throws Exception {
        Person worker = person();
        Project project = project(worker);
        grant(worker.id(), project.id(), TICKETS);
        grant(worker.id(), project.id(), "tag.create");
        long number = ticket(worker, project, "Last");
        List<UUID> tags = new ArrayList<>();
        for (int i = 0; i < 8; i++) {
            tags.add(tag(worker, catalog(project), "Last " + i));
        }
        String path = "/api/v1/projects/" + project.id() + "/tickets/" + number;
        long deadlocksBefore = jdbc.queryForObject("select deadlocks from pg_stat_database where datname = current_database()", Long.class);
        List<Future<List<Integer>>> work = new ArrayList<>();
        for (int i = 0; i < 8; i++) {
            UUID tag = tags.get(i);
            int index = i;
            work.add(POOL.submit(() -> {
                List<Integer> statuses = new ArrayList<>();
                for (int round = 0; round < 3; round++) {
                    statuses.add(post(worker, ticketTags(project, number), Map.of("tagId", tag.toString())).status());
                    statuses.add(delete(worker, ticketTags(project, number) + "/" + tag).status());
                }
                return statuses;
            }));
            work.add(POOL.submit(() -> {
                List<Integer> statuses = new ArrayList<>();
                for (int round = 0; round < 3; round++) {
                    while (true) {
                        String etag = worker.get(path).header("ETag");
                        int status = patch(worker, path, Map.of("title", "T" + index + "-" + round), etag).status();
                        if (status != 412) {
                            statuses.add(status);
                            break;
                        }
                    }
                }
                return statuses;
            }));
        }
        List<Integer> all = new ArrayList<>();
        for (Future<List<Integer>> future : work) {
            all.addAll(future.get(120, TimeUnit.SECONDS));
        }
        long deadlocks = jdbc.queryForObject("select deadlocks from pg_stat_database where datname = current_database()", Long.class) - deadlocksBefore;
        assertEquals(0, deadlocks, "deadlocks");
        assertTrue(all.stream().allMatch(s -> s == 200 || s == 204), "statuses " + all);

        Response seen = worker.get(path + "/history");
        long last = 0;
        int tagEntries = 0;
        for (JsonNode entry : seen.json()) {
            long version = entry.get("version").asLong();
            assertTrue(version >= last, "versions must not go back: " + lines(seen));
            last = version;
            if (entry.get("changes").get(0).get("field").asString().equals("tags")) {
                tagEntries++;
            }
        }
        assertEquals(48, tagEntries, "one entry per actual assignment and removal");
        assertEquals("\"" + last + "\"", worker.get(path).header("ETag"));
    }
