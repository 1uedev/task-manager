    @Test
    void probeOldFetchCreatesSecondTicketAfterNewGenerationAlreadyImportedSameMail() throws Exception {
        Setup s=setup();
        jdbc.update("update helpdesk_mailbox set uid_validity=100,last_uid=0 where id=?",s.mailbox());
        byte[] same=scripted(s,"same-message","");
        var entered=new java.util.concurrent.CountDownLatch(1);
        var release=new java.util.concurrent.CountDownLatch(1);
        ScriptedMailbox old=new ScriptedMailbox(100).put(100,same)
            .beforeFetch(uid->{entered.countDown();hold(release);});
        Future<?> a=POOL.submit(()->old.intake(context,1).fetch(s.mailbox()));
        try {
            hold(entered);
            new ScriptedMailbox(200).put(1,same).intake(context,1).fetch(s.mailbox());
            assertEquals(1,tickets(s));
        } finally {release.countDown();}
        a.get(20,java.util.concurrent.TimeUnit.SECONDS);
        assertEquals(2,tickets(s),"observed defect: stale fetch creates duplicate ticket");
        assertEquals(1L,jdbc.queryForObject("select count(distinct message_id) from helpdesk_message where mailbox_id=?",Long.class,s.mailbox()));
        assertEquals(2,jdbc.queryForObject("select source from helpdesk_mailbox where id=?",Integer.class,s.mailbox()));
        assertEquals(1L,jdbc.queryForObject("select last_uid from helpdesk_mailbox where id=?",Long.class,s.mailbox()));
        System.out.println("E1 observed: same Message-ID, two sources, two tickets; current cursor preserved");
    }

        ScriptedMailbox old=new ScriptedMailbox(100).put(99,scripted(next,"old-settings",""))

    }
