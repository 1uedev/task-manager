    @Test
    void reviewBusyAndConcurrentQueueActions() throws Exception {
        Setup s=setup();
        Browser reviewer=reviewer(s);
        deliver(s,"","Queue race","Synthetic"); intake.fetch(s.mailbox());
        UUID id=(UUID)messages(s).get(0).get("id");
        String path="/api/v1/helpdesk/inbox/"+id;
        String etag=reviewer.get(path).etag();
        try(var connection=jdbc.getDataSource().getConnection()) {
            connection.setAutoCommit(false);
            try(var statement=connection.prepareStatement("select id from helpdesk_message where id=? for update")) {
                statement.setObject(1,id); statement.executeQuery().close();
                Response busy=reviewer.send("POST",path+"/discard",Map.of("reason","Synthetic"),etag);
                assertEquals(503,busy.status(),busy.body()); assertEquals("helpdesk.busy",busy.code());
                assertEquals("review",messages(s).get(0).get("state"));
            } finally {connection.rollback();}
        }
        var start=new java.util.concurrent.CountDownLatch(1);
        Future<Response> ticket=POOL.submit(()->{start.await();return reviewer.send("POST",path+"/ticket",Map.of(),etag);});
        Future<Response> discard=POOL.submit(()->{start.await();return reviewer.send("POST",path+"/discard",Map.of("reason","Synthetic"),etag);});
        start.countDown();
        List<Integer> statuses=new ArrayList<>(List.of(ticket.get().status(),discard.get().status()));
        statuses.sort(Integer::compareTo); assertEquals(List.of(200,409),statuses);
        assertTrue(tickets(s)<=1);
        System.out.println("E8 queue: held lock returns 503 helpdesk.busy without mutation; concurrent ticket/discard returns 200/409");
    }

    @Test
    void reviewImapFailureLeaksReflectedSecretIntoLog() throws Exception {
        Setup s=setup();
        var capture=new ch.qos.logback.core.read.ListAppender<ch.qos.logback.classic.spi.ILoggingEvent>();
        capture.start();
        var log=(ch.qos.logback.classic.Logger)org.slf4j.LoggerFactory.getLogger("taskmanager.helpdesk.internal");
        log.addAppender(capture);
        try(ServerSocket server=new ServerSocket(0)) {
            server.setSoTimeout(15000);
            var peer=Executors.newSingleThreadExecutor();
            var finished=peer.submit(() -> {
                try {
                    for(int n=0;n<2;n++) try(var socket=server.accept()) {
                        socket.setSoTimeout(10000);
                        var reader=new java.io.BufferedReader(new java.io.InputStreamReader(socket.getInputStream()));
                        var writer=new java.io.PrintWriter(socket.getOutputStream(),true);
                        writer.print("* OK synthetic IMAP\r\n"); writer.flush();
                        for(String line;(line=reader.readLine())!=null;) {
                            String tag=line.split(" ")[0];
                            if(line.toUpperCase().contains("CAPABILITY")) {
                                writer.print("* CAPABILITY IMAP4rev1\r\n"+tag+" OK capabilities\r\n"); writer.flush();
                            } else if(line.toUpperCase().contains("LOGIN")) {
                                writer.print(tag+" NO rejected "+MAIL_PASSWORD+"\r\n"); writer.flush(); break;
                            } else {writer.print(tag+" OK\r\n");writer.flush();}
                        }
                    }
                } catch(Exception failure) {throw new RuntimeException(failure);}
            });
            try {
                jdbc.update("update helpdesk_mailbox set imap_port=? where id=?",server.getLocalPort(),s.mailbox());
                Response result=s.admin().send("POST","/api/v1/helpdesk/mailboxes/"+s.mailbox()+"/test",null,null);
                assertEquals("helpdesk.authentication_failed",result.json().get("code").asString());
                assertFalse(result.body().contains(MAIL_PASSWORD));
                intake.fetch(s.mailbox());
                finished.get(20,TimeUnit.SECONDS);
                assertEquals(2,capture.list.stream().filter(e->e.getFormattedMessage().contains(MAIL_PASSWORD)).count());
                System.out.println("E7 IMAP failure: reflected synthetic password appears in both test and fetch logs; API clean");
            } finally {peer.shutdownNow();}
        } finally {log.detachAppender(capture);capture.stop();}
    }
